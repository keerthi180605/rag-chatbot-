<script>
const storyLines = [
  'You held my hand before I knew how to walk… 🤍',
  'You believed in me even when I doubted myself 🌷',
  'Your love never asked for anything in return ✨',
  'Everything I am today begins with you, Amma 💖'
];
let index = 0;

function startJourney(){
  document.getElementById('lock').style.display='none';
  document.getElementById('journey').classList.remove('hidden');
  heartFall();
  setTimeout(()=>{
    document.getElementById('controls').classList.remove('hidden');
  },3000);
}

function nextStory(){
  if(index < storyLines.length){
    document.getElementById('story').innerText = storyLines[index];
    index++;
  }
  if(index === storyLines.length){
    document.getElementById('controls').style.display='none';
    document.getElementById('gratitude').classList.remove('hidden');
  }
}

function showPhoto(){
  setTimeout(()=>{
    document.getElementById('photo').style.display='block';
    document.getElementById('ending').classList.remove('hidden');
  },1500);
}

function heartFall(){
  setInterval(()=>{
    const h=document.createElement('div');
    h.className='heart';
    h.innerHTML='🌸';
    h.style.left=Math.random()*100+'vw';
    h.style.animationDuration=6+Math.random()*4+'s';
    document.body.appendChild(h);
    setTimeout(()=>h.remove(),10000);
  },800);
}
</script>
